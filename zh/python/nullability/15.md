# --solutions--

```python
def oldest(people):
    best_name = None
    best_age = None
    for person in people:
        age = person.get("age")
        if age is None:
            continue
        if best_age is None or age > best_age:
            best_age = age
            best_name = person["name"]
    return best_name
```
