# --description--

在**尾部**添加节点需要先找到最后一个节点，即 `next` 为 `NULL` 的那个节点。函数一路走到它，把新节点接在那里，并返回未改变的头部：
```c
Node *last = head;
while (last->next != NULL) {
    last = last->next;
}
last->next = node;
```
链表为空时没有可以走到的最后一个节点：新节点直接作为头节点返回。

# --solutions--

```c
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int value; struct Node *next; } Node;
Node *push_back(Node *head, int value) {
    Node *node = malloc(sizeof(Node));
    node->value = value;
    node->next = NULL;
    if (head == NULL) {
        return node;
    }
    Node *last = head;
    while (last->next != NULL) {
        last = last->next;
    }
    last->next = node;
    return head;
}
int main() {
    Node *head = NULL;
    head = push_back(head, 1);
    head = push_back(head, 2);
    head = push_back(head, 3);
    for (Node *n = head; n != NULL; n = n->next) {
        printf("%d\n", n->value);
    }
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
    return 0;
}
```
