```swift
protocol Badge {
    var symbol: String { get }
}
extension Badge {
    func banner() -> String { return "[\(symbol)]" }
}
enum Rank: Badge {
    case gold

    var symbol: String { return "G" }
}
print(Rank.gold.banner())
```
