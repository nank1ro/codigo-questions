# --description--

Блок `do` не ограничен одним типом ошибки. Каждый шаг может завершиться неудачей по-своему, и каждый сбой получает собственный `catch`:
```swift
do {
    let text = try load(false)
    let value = try parse(text)
    print(value)
} catch NetworkError.offline {
    print("offline")
} catch ParseError.badFormat {
    print("bad format")
} catch {
    print("unknown")
}
```
Первый `try`, выбросивший ошибку, завершает блок, поэтому последующие шаги уже не выполняются — значение просто никогда не существовало. Именно это делает такую форму читаемой: успешный путь остаётся одной прямой строкой сверху, а все способы, которыми всё может пойти не так, перечислены под ним.

enum NetworkError: Error {
    case offline
}

enum ParseError: Error {
    case badFormat
}

func load(_ online: Bool) throws -> String {
    guard online else {
        throw NetworkError.offline
    }
    return "raw"
}

func parse(_ text: String) throws -> Int {
    throw ParseError.badFormat
}

do {
    let text = try load(false)
    let value = try parse(text)
    print(value)
} catch [/] {
    print("offline")
} catch [/] {
    print("bad format")
} [/] {
