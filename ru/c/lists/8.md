# --description--

Добавление узла в **конец** требует последний узел — тот, чей `next` равен `NULL`. Функция идёт по списку, пока не найдёт его, затем подключает новый узел там и возвращает неизменённую голову:
```c
Node *last = head;
while (last->next != NULL) {
    last = last->next;
}
last->next = node;
```
Когда список пуст, идти до последнего узла не нужно: новый узел просто возвращается как голова.

# --solutions--

```c
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int value; struct Node *next; } Node;
Node *push_back(Node *head, int value) {
    Node *node = malloc(sizeof(Node));
    node->value = value;
    node->next = NULL;
    if (head == NULL) {
        return node;
    }
    Node *last = head;
    while (last->next != NULL) {
        last = last->next;
    }
    last->next = node;
    return head;
}
int main() {
    Node *head = NULL;
    head = push_back(head, 1);
    head = push_back(head, 2);
    head = push_back(head, 3);
    for (Node *n = head; n != NULL; n = n->next) {
        printf("%d\n", n->value);
    }
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
    return 0;
}
```
