# --description--

Aggiungere un nodo **in fondo** richiede l'ultimo nodo, quello il cui `next` è `NULL`. La funzione cammina finché non lo trova, poi vi aggancia il nuovo nodo e restituisce la testa invariata:
```c
Node *last = head;
while (last->next != NULL) {
    last = last->next;
}
last->next = node;
```
Quando la lista è vuota non c'è un ultimo nodo fino a cui camminare: il nuovo nodo è semplicemente restituito come testa.

# --solutions--

```c
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int value; struct Node *next; } Node;
Node *push_back(Node *head, int value) {
    Node *node = malloc(sizeof(Node));
    node->value = value;
    node->next = NULL;
    if (head == NULL) {
        return node;
    }
    Node *last = head;
    while (last->next != NULL) {
        last = last->next;
    }
    last->next = node;
    return head;
}
int main() {
    Node *head = NULL;
    head = push_back(head, 1);
    head = push_back(head, 2);
    head = push_back(head, 3);
    for (Node *n = head; n != NULL; n = n->next) {
        printf("%d\n", n->value);
    }
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
    return 0;
}
```
