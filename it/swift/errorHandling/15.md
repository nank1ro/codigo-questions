```swift
enum AgeError: Error {
    case negative
}
func label(_ age: Int) throws -> String {
    if age < 0 {
        throw AgeError.negative
    }
    return "age \(age)"
}
for age in [4, -1, 7] {
    do {
        print(try label(age))
    } catch {
        print("skipped")
    }
}
```
