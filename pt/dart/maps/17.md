```dart
void main() {
  var counts = <String, int>{};
  for (var word in ['a', 'b', 'a']) {
    counts.update(word, (n) => n + 1, ifAbsent: () => 1);
  }
  print(counts);
}
```

# --output--

{a: 2, b: 1}
