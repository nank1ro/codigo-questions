```c
void print_person(Person p) {
    printf("%s (%d)\n", p.name, p.age);
}
```

# --solutions--

```c
#include <stdio.h>
typedef struct { int x; int y; } Point;
void print_point(Point p) {
    printf("(%d, %d)\n", p.x, p.y);
}
int main() {
    Point p = {3, 4};
    print_point(p);
    return 0;
}
```

# --output--

(3, 4)
