```kotlin
sealed class Payment
data class Card(val digits: String) : Payment()
data class Transfer(val iban: String) : Payment()

fun label(payment: Payment): String = when (payment) {
    is Card -> "card ${payment.digits}"
    is Transfer -> "transfer ${payment.iban}"
}

fun main() {
    println(label(Transfer("IT60X")))
}
```
