```dart
enum Planet {
  mercury(0),
  earth(1),
  mars(2);
  final int moons;
  const Planet(this.moons);
}
void main() {
  for (var planet in Planet.values) {
    print('${planet.name}: ${planet.moons}');
  }
}
```
