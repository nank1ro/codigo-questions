# --description--

Ajouter un nœud **à la fin** demande le dernier nœud, celui dont le `next` est `NULL`. La fonction avance jusqu'à le trouver, puis y attache le nouveau nœud et renvoie la tête inchangée :
```c
Node *last = head;
while (last->next != NULL) {
    last = last->next;
}
last->next = node;
```
Lorsque la liste est vide, il n'y a pas de dernier nœud où avancer : le nouveau nœud est simplement renvoyé comme tête.

# --solutions--

```c
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int value; struct Node *next; } Node;
Node *push_back(Node *head, int value) {
    Node *node = malloc(sizeof(Node));
    node->value = value;
    node->next = NULL;
    if (head == NULL) {
        return node;
    }
    Node *last = head;
    while (last->next != NULL) {
        last = last->next;
    }
    last->next = node;
    return head;
}
int main() {
    Node *head = NULL;
    head = push_back(head, 1);
    head = push_back(head, 2);
    head = push_back(head, 3);
    for (Node *n = head; n != NULL; n = n->next) {
        printf("%d\n", n->value);
    }
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
    return 0;
}
```
