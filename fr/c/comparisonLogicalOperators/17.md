# --solutions--

```c
#include <stdio.h>
#include <stdbool.h>
bool can_vote(int age, bool citizen) {
    return age >= 18 && citizen;
}
int main() {
    printf("%d\n", can_vote(20, true));
    printf("%d\n", can_vote(20, false));
    return 0;
}
```
