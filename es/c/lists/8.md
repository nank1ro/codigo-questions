# --description--

Añadir un nodo al **final** necesita el último nodo, el que tiene `next` igual a `NULL`. La función avanza hasta encontrarlo, luego engancha el nodo nuevo ahí y devuelve la cabeza sin cambios:
```c
Node *last = head;
while (last->next != NULL) {
    last = last->next;
}
last->next = node;
```
Cuando la lista está vacía no hay último nodo al que llegar: el nodo nuevo simplemente se devuelve como la cabeza.

# --solutions--

```c
#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int value; struct Node *next; } Node;
Node *push_back(Node *head, int value) {
    Node *node = malloc(sizeof(Node));
    node->value = value;
    node->next = NULL;
    if (head == NULL) {
        return node;
    }
    Node *last = head;
    while (last->next != NULL) {
        last = last->next;
    }
    last->next = node;
    return head;
}
int main() {
    Node *head = NULL;
    head = push_back(head, 1);
    head = push_back(head, 2);
    head = push_back(head, 3);
    for (Node *n = head; n != NULL; n = n->next) {
        printf("%d\n", n->value);
    }
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
    return 0;
}
```
