```c
typedef struct { int x; int y; } Point;
Point path[3] = {{0, 0}, {2, 1}, {4, 3}};
int sum = 0;
for (int i = 0; i < 3; i++) {
    sum += path[i].x;
}
printf("%d\n", sum);
```

# --output--

6
