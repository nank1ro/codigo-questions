```dart
T smallest<T extends Comparable<T>>(List<T> items) {
  var best = items.first;
  for (final item in items) {
    if (item.compareTo(best) < 0) best = item;
  }
  return best;
}

void main() {
  print(smallest(['pear', 'fig', 'kiwi']));
}
```
